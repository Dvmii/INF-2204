We ran this in python:

'''
import pandas as pd
import dabest

df = pd.read_csv("dabest_cliff_632_vs_640.csv") 

analysis = dabest.load(
    data=df,
    idx=("632 MiB", "640 MiB"),
    paired="baseline",
    id_col="repeat",
    ci=95,
    resamples=20000,
    random_seed=2204
)

print(dabest.__version__) 
print(analysis.mean_diff)

print(
    analysis.mean_diff.statistical_tests[
        ["difference", "bca_low", "bca_high", "pvalue_wilcoxon"]
    ]
)
'''
