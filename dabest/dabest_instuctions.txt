import pandas as pd
import dabest

df = pd.read_csv("dabest_cliff_632_vs_640.csv")

    data=df,
    idx=("632 MiB", "640 MiB"),
    paired="baseline",
    id_col="repeat",
    ci=95,
    resamples=20000,
    random_seed=2204
)

print(dabest.__version__) 
print(analysis.mean_diff)

)

We also ran this in python:

'''
import dabest
import numpy as np
import pandas as pd
from dabest._stats_tools.confint_1group import compute_1group_bootstraps

df = pd.read_csv("dabest_speedup_input.csv")
threads = [1, 2, 4, 8, 16]
x = df[["t1", "t2", "t4", "t8", "t16"]].to_numpy()

assert df["repeat"].is_unique
assert x.shape == (40, 5) and np.isfinite(x).all()

speedup = lambda a: np.median(a, axis=0)[0] / np.median(a, axis=0)

bootstraps = compute_1group_bootstraps(
    x, speedup, resamples=20000, random_seed=2204
)

low, high = np.quantile(bootstraps, [0.025, 0.975], axis=0)
estimate = speedup(x)

result = pd.DataFrame({
    "threads": threads,
    "speedup": estimate,
    "ci_low": low,
    "ci_high": high,
    "error_minus": estimate - low,
    "error_plus": high - estimate,
})

print("DABEST:", dabest.__version__)
print(result.round(6).to_string(index=False))
result.to_csv("speedup_dabest_results.csv", index=False)
'''
